        function renderMotorActivities() {
            const container = document.getElementById('motor-list');
            const search = document.getElementById('motor-search').value.toLowerCase();
            container.innerHTML = '';
            
            const filtered = motorActivities.filter(act => act.toLowerCase().includes(search));
            
            if (filtered.length === 0) {
                container.innerHTML = '<p class="text-sm text-slate-400 p-2">No se encontraron actividades.</p>';
                return;
            }
            filtered.forEach(act => {
                const div = document.createElement('div');
                div.className = 'bg-white p-2 rounded border border-orange-100 text-sm text-slate-700 flex gap-2 items-start';
                div.innerHTML = `<i class="fa-solid fa-check text-orange-400 mt-1"></i> <span>${act}</span>`;
                container.appendChild(div);
            });

        // --- JUMP FUNCTION ---
        function jumpToBank(targetId) {
            if(!targetId) return;
            switchTab('banco');
            
            // Wait for tab switch
            setTimeout(() => {
                const element = document.getElementById(targetId);
                if(element) {
                    element.scrollIntoView({ behavior: 'smooth', block: 'center' });
                    element.classList.add('highlight-pulse');
                    setTimeout(() => {
                        element.classList.remove('highlight-pulse');
                    }, 4000);
                }
            }, 100);
        }

        // --- TOGGLE DETALLES ---
        function toggleDetails(index) {
            const content = document.getElementById(`details-${index}`);
            const arrow = document.getElementById(`arrow-${index}`);
            
            if (content.classList.contains('details-open')) {
                content.classList.remove('details-open');
                arrow.style.transform = 'rotate(0deg)';
            } else {
                content.classList.add('details-open');
                arrow.style.transform = 'rotate(180deg)';
            }
        }

        // --- LÓGICA DE PROGRESO ---
        function toggleDay(index, checkbox) {
            const isChecked = checkbox.checked;
            localStorage.setItem(`zaid_day_${index}`, isChecked);
            
            if (isChecked) {
                confetti({
                    particleCount: 100,
                    spread: 70,
                    origin: { y: 0.6 },
                    colors: ['#3b82f6', '#10b981', '#f59e0b']
                });
            }
            setTimeout(() => { renderSchedule(); }, 300);
        }

        function updateProgress() {
            let completed = 0;
            for (let i = 0; i < 7; i++) {
                if (localStorage.getItem(`zaid_day_${i}`) === 'true') completed++;
            }
            document.getElementById('progress-text').textContent = `${completed}/7 Días`;
            
            // Update Bar
            const pct = (completed / 7) * 100;
            document.getElementById('progress-bar').style.width = `${pct}%`;

            // Button Logic
            const finishBtn = document.getElementById('btn-finish-week');
            
            if (completed === 7) {
                // Unlock state
                finishBtn.disabled = false;
                finishBtn.classList.remove('bg-gray-300', 'text-gray-500', 'cursor-not-allowed', 'shadow-none', 'border-gray-400');
                finishBtn.classList.add('bg-yellow-400', 'text-blue-900', 'hover:bg-yellow-300', 'shadow-md', 'border-yellow-600', 'active:border-b-0', 'active:translate-y-1', 'border-b-4');
                finishBtn.innerHTML = '<i class="fa-solid fa-trophy mr-1"></i> ¡Completar Semana!';
            } else {
                // Locked state
                finishBtn.disabled = true;
                finishBtn.classList.add('bg-gray-300', 'text-gray-500', 'cursor-not-allowed', 'shadow-none', 'border-gray-400');
                finishBtn.classList.remove('bg-yellow-400', 'text-blue-900', 'hover:bg-yellow-300', 'shadow-md', 'border-yellow-600', 'active:border-b-0', 'active:translate-y-1');
                 finishBtn.innerHTML = `<i class="fa-solid fa-lock mr-2"></i> ${completed}/7`;
            }
        }

        function finishWeek() {
            let currentLevel = parseInt(localStorage.getItem('zaid_level') || '1');
            currentLevel++;
            localStorage.setItem('zaid_level', currentLevel);
            
            // Big Celebration
            var end = Date.now() + (2 * 1000);
            var colors = ['#bb0000', '#ffffff'];
            (function frame() {
              confetti({
                particleCount: 2,
                angle: 60,
                spread: 55,
                origin: { x: 0 },
                colors: colors
              });
              confetti({
                particleCount: 2,
                angle: 120,
                spread: 55,
                origin: { x: 1 },
                colors: colors
              });
              if (Date.now() < end) {
                requestAnimationFrame(frame);
              }
            }());

            // Clear days
            for(let i=0; i<7; i++) localStorage.removeItem(`zaid_day_${i}`);
            
            alert(`¡Felicidades Equipo! Han subido al Nivel ${currentLevel}. \nAhora el cronograma se ha reiniciado. ¡Usen el Banco de Ideas para nuevas actividades esta semana!`);
            
            renderSchedule();
        }

        function loadStreak() {
            let currentLevel = localStorage.getItem('zaid_level') || '1';
            document.getElementById('level-badge').textContent = `🏆 Nivel ${currentLevel}`;
        }

        function resetProgress() {
            if(confirm("¿Seguro? Esto borrará el progreso de ESTA semana (no el Nivel).")) {
                for(let i=0; i<7; i++) localStorage.removeItem(`zaid_day_${i}`);
                renderSchedule();
            }
        }

        // --- NAVEGACIÓN ---
        function switchTab(tabId) {
            document.querySelectorAll('.tab-content').forEach(el => {
                el.classList.add('hidden');
                el.classList.remove('block', 'animate-fade-in');
            });
            
            const selected = document.getElementById(tabId);
            selected.classList.remove('hidden');
            selected.classList.add('block', 'animate-fade-in');

            document.querySelectorAll('.nav-btn').forEach(btn => {
                btn.classList.remove('nav-active', 'bg-white', 'text-blue-600', 'shadow-md');
                btn.classList.add('bg-blue-100', 'text-blue-600');
            });
            const activeBtn = document.getElementById(`btn-${tabId}`);
            if (activeBtn) {
                activeBtn.classList.add('nav-active');
                activeBtn.classList.remove('bg-blue-100');
            }
        }
